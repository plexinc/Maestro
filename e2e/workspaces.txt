# The workspaces the e2e suite knows about, and what each one supports.
#
#   name       MAESTRO_APP value, and the name used for debug-output directories
#   path       flow directory, relative to e2e/
#   platforms  where it runs: android, ios, web, tvos or vega, or several joined by commas with no
#              spaces, e.g. android,ios
#   failing    yes when it carries `failing`-tagged flows, which are expected to fail
#              and so are run with inverted polarity
#
# Order matters only in that it is the order the full suite runs them in.
#
# run_tests validates this file before doing anything, so a mistake here is reported
# against the line rather than quietly dropping a workspace from the run.
#
# name             path                        platforms  failing

# Skipped on ios: the Flutter demo app hits OOM on GHA's macOS runners.
demo_app           demo_app/.maestro           android    yes

# Prebuilt apps downloaded from manifest.txt. Both are iOS-only here, and neither
# carries failing flows.
simple_web_view    workspaces/simple_web_view  ios        no
wikipedia          workspaces/wikipedia        ios        no

# Browser flows: the only workspace with no app behind it, so there is nothing to install
# and nothing that could fail on a device.
web                workspaces/web              web        no

# Apple TV: need a tvOS simulator, so they run only on tvos.
tvos_settings      workspaces/tvos_settings    tvos       no
tvos_demo_app      workspaces/tvos_demo_app    tvos       no

# Amazon Vega (Fire TV): needs a Vega device, so it runs only on vega.
vega_demo_app      workspaces/vega_demo_app    vega       no
